  --dataset_config DATASET_CONFIG

  --batch_size BATCH_SIZE
                        batch size, override dataset config if dataset batch
                        size > this
  --num_workers NUM_WORKERS
                        number of workers for dataset. default is cpu count-1
  --skip_existing       skip existing cache files
  --keep_cache          keep cache files not in dataset
  --model_type {full,dev}
                        HiDream-O1 model variant
  --dit DIT             HiDream-O1 single checkpoint for optional cached text
                        embeddings
  --fp8_te              Cache HiDream-O1 text token embeddings in fp8
                        precision
